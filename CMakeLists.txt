cmake_minimum_required(VERSION 3.24)
project(tidal VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(TIDAL_CUDA "Build the strict CUDA execution path" OFF)
option(TIDAL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
set(TIDAL_GGML_SOURCE_DIR "" CACHE PATH "Use an existing checkout of the pinned ggml revision")

set(GGML_CUDA ${TIDAL_CUDA} CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
if(TIDAL_GGML_SOURCE_DIR)
    add_subdirectory("${TIDAL_GGML_SOURCE_DIR}" ggml)
else()
    include(FetchContent)
    FetchContent_Declare(ggml
        GIT_REPOSITORY https://github.com/ggml-org/ggml.git
        GIT_TAG facdc44a67da17df4433adcc468b424f48e9ded2)
    FetchContent_MakeAvailable(ggml)
endif()

add_library(tidal src/model.cpp src/engine.cpp src/sparse_attention.cpp)
target_include_directories(tidal PUBLIC include PRIVATE src)
target_link_libraries(tidal PUBLIC ggml ggml-cpu)
if(TIDAL_CUDA)
    target_link_libraries(tidal PRIVATE ggml-cuda)
    target_compile_definitions(tidal PRIVATE TIDAL_CUDA=1)
endif()
if(MSVC)
    target_compile_options(tidal PRIVATE /W4)
else()
    target_compile_options(tidal PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TIDAL_SANITIZE AND NOT MSVC)
    target_compile_options(tidal PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tidal PUBLIC -fsanitize=address,undefined)
endif()

add_executable(tidal-cli tools/main.cpp)
target_link_libraries(tidal-cli PRIVATE tidal)
add_executable(tidal-dummy tools/dummy.cpp)
target_link_libraries(tidal-dummy PRIVATE tidal)
target_include_directories(tidal-dummy PRIVATE src)

include(CTest)
if(BUILD_TESTING)
    add_executable(tidal-tests tests/test_core.cpp)
    target_link_libraries(tidal-tests PRIVATE tidal)
    target_include_directories(tidal-tests PRIVATE src)
    add_test(NAME core COMMAND tidal-tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy"
            RESULT_VARIABLE TIDAL_NUMPY_STATUS OUTPUT_QUIET ERROR_QUIET)
        if(TIDAL_NUMPY_STATUS EQUAL 0)
            add_test(NAME numerical-reference COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference.py
                --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(numerical-reference PROPERTIES TIMEOUT 180)
        else()
            message(STATUS "NumPy unavailable: numerical-reference test not registered")
        endif()
    endif()
endif()
