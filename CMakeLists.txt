cmake_minimum_required(VERSION 3.24)
project(tidal VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(TIDAL_CUDA "Build the strict CUDA execution path" OFF)
option(TIDAL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
set(TIDAL_GGML_SOURCE_DIR "" CACHE PATH "Use an existing checkout of the pinned ggml revision")

set(GGML_CUDA ${TIDAL_CUDA} CACHE BOOL "" FORCE)
if(TIDAL_CUDA)
    set(GGML_CUDA_FA ON CACHE BOOL "Required by GPU layer prefill" FORCE)
endif()
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
if(TIDAL_GGML_SOURCE_DIR)
    add_subdirectory("${TIDAL_GGML_SOURCE_DIR}" ggml)
    set(TIDAL_GGML_BUILD_SOURCE "${TIDAL_GGML_SOURCE_DIR}")
else()
    include(FetchContent)
    FetchContent_Declare(ggml
        GIT_REPOSITORY https://github.com/ggml-org/ggml.git
        GIT_TAG facdc44a67da17df4433adcc468b424f48e9ded2)
    FetchContent_MakeAvailable(ggml)
    set(TIDAL_GGML_BUILD_SOURCE "${ggml_SOURCE_DIR}")
endif()

add_library(tidal src/model.cpp src/qwen35.cpp src/quant.cpp src/ple.cpp src/engine.cpp src/sparse_attention.cpp src/sampling.cpp src/prefill_memory.cpp src/kv_export.cpp)
target_include_directories(tidal PUBLIC include PRIVATE src)
target_link_libraries(tidal PUBLIC ggml ggml-cpu)
if(TIDAL_CUDA)
    enable_language(CUDA)
    target_sources(tidal PRIVATE src/sampling_cuda.cu)
    set_target_properties(tidal PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(tidal PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
    target_link_libraries(tidal PRIVATE ggml-cuda)
    target_compile_definitions(tidal PRIVATE TIDAL_CUDA=1)
    find_package(CUDAToolkit REQUIRED)
    find_path(TIDAL_NVTX_INCLUDE nvtx3/nvToolsExt.h HINTS ${CUDAToolkit_INCLUDE_DIRS})
    if(TIDAL_NVTX_INCLUDE)
        target_include_directories(tidal PRIVATE ${TIDAL_NVTX_INCLUDE})
        target_link_libraries(tidal PRIVATE ${CMAKE_DL_LIBS})
        target_compile_definitions(tidal PRIVATE TIDAL_NVTX=1)
    endif()
endif()
if(MSVC)
    target_compile_options(tidal PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
else()
    target_compile_options(tidal PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(TIDAL_SANITIZE AND NOT MSVC)
    target_compile_options(tidal PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>)
    target_link_options(tidal PUBLIC -fsanitize=address,undefined)
endif()

add_executable(tidal-cli tools/main.cpp)
target_link_libraries(tidal-cli PRIVATE tidal)
add_executable(tidal-dummy tools/dummy.cpp)
target_link_libraries(tidal-dummy PRIVATE tidal)
target_include_directories(tidal-dummy PRIVATE src)

add_executable(tidal-bench tools/bench.cpp)
target_link_libraries(tidal-bench PRIVATE tidal)
add_custom_target(tidal-build-info
    COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_SOURCE_DIR}
        -DGGML_DIR=${TIDAL_GGML_BUILD_SOURCE}
        -DOUTPUT=${CMAKE_BINARY_DIR}/tidal-build-info.hpp -P ${CMAKE_SOURCE_DIR}/cmake/build_info.cmake
    BYPRODUCTS ${CMAKE_BINARY_DIR}/tidal-build-info.hpp
    VERBATIM)
add_dependencies(tidal-bench tidal-build-info)
target_include_directories(tidal-bench PRIVATE ${CMAKE_BINARY_DIR})
if(TIDAL_CUDA)
    target_link_libraries(tidal-bench PRIVATE CUDA::cudart)
    target_compile_definitions(tidal-bench PRIVATE TIDAL_BENCH_CUDA=1)
endif()
target_compile_definitions(tidal-bench PRIVATE
    TIDAL_COMPILER="${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
    TIDAL_BUILD_TYPE="${CMAKE_BUILD_TYPE}")

include(CTest)
if(BUILD_TESTING)
    add_executable(tidal-prefill-test tests/prefill.cpp)
    target_link_libraries(tidal-prefill-test PRIVATE tidal)
    add_executable(tidal-residency-test tests/residency.cpp)
    target_link_libraries(tidal-residency-test PRIVATE tidal)
    target_include_directories(tidal-residency-test PRIVATE src)
    if(TIDAL_CUDA)
        target_link_libraries(tidal-residency-test PRIVATE ggml-cuda)
        target_compile_definitions(tidal-residency-test PRIVATE TIDAL_CUDA=1)
        if(TIDAL_NVTX_INCLUDE)
            target_include_directories(tidal-residency-test PRIVATE ${TIDAL_NVTX_INCLUDE})
            target_compile_definitions(tidal-residency-test PRIVATE TIDAL_NVTX=1)
        endif()
    endif()
    add_executable(tidal-sampling-test tests/sampling.cpp)
    target_link_libraries(tidal-sampling-test PRIVATE tidal)
    target_include_directories(tidal-sampling-test PRIVATE src tools)
    if(TIDAL_CUDA)
        target_link_libraries(tidal-sampling-test PRIVATE ggml-cuda)
        target_compile_definitions(tidal-sampling-test PRIVATE TIDAL_CUDA=1)
        if(TIDAL_NVTX_INCLUDE)
            target_include_directories(tidal-sampling-test PRIVATE ${TIDAL_NVTX_INCLUDE})
            target_compile_definitions(tidal-sampling-test PRIVATE TIDAL_NVTX=1)
        endif()
    endif()
    add_executable(tidal-weights-test tests/weights.cpp)
    target_link_libraries(tidal-weights-test PRIVATE tidal)
    target_include_directories(tidal-weights-test PRIVATE src)
    if(TIDAL_CUDA)
        target_link_libraries(tidal-weights-test PRIVATE ggml-cuda)
        target_compile_definitions(tidal-weights-test PRIVATE TIDAL_CUDA=1)
    endif()
    add_executable(tidal-state-test tests/state.cpp)
    target_link_libraries(tidal-state-test PRIVATE tidal)
    add_executable(tidal-matvec-test tests/matvec.cpp)
    target_link_libraries(tidal-matvec-test PRIVATE tidal)
    target_include_directories(tidal-matvec-test PRIVATE src)
    if(TIDAL_CUDA)
        target_link_libraries(tidal-matvec-test PRIVATE ggml-cuda)
        target_compile_definitions(tidal-matvec-test PRIVATE TIDAL_CUDA=1)
    endif()
    add_executable(tidal-tests tests/test_core.cpp)
    target_link_libraries(tidal-tests PRIVATE tidal)
    target_include_directories(tidal-tests PRIVATE src)
    add_test(NAME core COMMAND tidal-tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME model-plan COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_plan.py --bin-dir $<TARGET_FILE_DIR:tidal-dummy>)
        add_test(NAME benchmark COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark.py
            --bin-dir $<TARGET_FILE_DIR:tidal-bench>)
        set_tests_properties(benchmark PROPERTIES TIMEOUT 180)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy"
            RESULT_VARIABLE TIDAL_NUMPY_STATUS OUTPUT_QUIET ERROR_QUIET)
        if(TIDAL_NUMPY_STATUS EQUAL 0)
            add_test(NAME prefill COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_prefill.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(prefill PROPERTIES TIMEOUT 240)
            add_test(NAME residency COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_residency.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(residency PROPERTIES TIMEOUT 180)
            add_test(NAME sampling COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sampling.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(sampling PROPERTIES TIMEOUT 180)
            add_test(NAME qwen35-reference COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_qwen35.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(qwen35-reference PROPERTIES TIMEOUT 180)
            add_test(NAME quantization COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_quant.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(quantization PROPERTIES TIMEOUT 600)
            add_test(NAME moe-reference COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_moe.py --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(moe-reference PROPERTIES TIMEOUT 180)
            add_test(NAME numerical-reference COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference.py
                --bin-dir $<TARGET_FILE_DIR:tidal-cli>)
            set_tests_properties(numerical-reference PROPERTIES TIMEOUT 180)
        else()
            message(STATUS "NumPy unavailable: numerical-reference test not registered")
        endif()
    endif()
endif()
